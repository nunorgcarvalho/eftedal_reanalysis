% ============================== COLUMN 1 ==============================
\posterbox[adjusted title=Background]{name=bg, column=1, below=header}{
\begin{itemize}
  \item \textbf{Eftedal et al.\ (2025, \textit{PNAS})}: test-score correlations for \textbf{82 categories} of
        relatives among $\sim$1M Norwegian schoolchildren: twins $\rightarrow$ 2nd cousins, in-laws, adoptees
  \item Fisherian model (additive genetics + assortative mating):
        $h^2 = 0.80$ in biological relatives, but \textbf{0.50} once in-laws are added
  \item Their conclusion: in-laws ``break'' the genetic model
  \item \textbf{We re-fit their public correlations} with corrected estimation and richer models
\end{itemize}
}

\posterbox[adjusted title=Model]{name=model, column=1, below=bg}{
Predicted correlation of relative category $k$:
\[
  \rho_k = V_A\,K_k + V_{\mathrm{GxG}}\,K_k^2 + V_{\mathrm{fam}}\, r_f^{\,f_k} \;\big[+\, V_{\mathrm{house}}\,h_k\big]
\]
\vspace{-0.8em}
\begin{itemize}
  \item $K_k$: additive genetic sharing under assortative mating (Fisher; Nagylaki), with \mbox{$m = h^2 r$}
  \item $V_{\mathrm{GxG}}$: pairwise gene--gene interaction
  \item $V_{\mathrm{fam}}\, r_f^{f}$: family environment decaying with family distance $f$
  \item Weighted least squares on all 82 correlations
\end{itemize}
}

\posterbox[adjusted title=What we fixed]{name=fixes, column=1, between=model and bottom}{
\begin{itemize}
  \item \textbf{Log-scale weights}: $\ln\rho$ was weighted by $1/SE^2$, but
        $\mathrm{Var}(\ln\hat\rho)\approx SE^2/\rho^2$ $\Rightarrow$ distant relatives over-weighted
        $\sim 1/\rho^2$ ($\times$200 for 2nd cousins). \textbf{We fit on the linear scale.}
  \item \textbf{$m$ and $r$} were estimated independently; the model implies $m = h^2 r$. \textbf{Coupled.}
  \item \textbf{Step-sibling covariance}: $m(\tfrac{1+m}{2})^2 \rightarrow m(\tfrac{1+r}{2})^2$
  \item \textbf{Weights}: inverse variance, and block-balanced (every kind of relative counts equally)
  \item \textbf{Uncertainty}: bootstrap + leave-one-block-out jackknife; models compared by
        \textbf{block cross-validation}
\end{itemize}
}

\posterbox[adjusted title=Conclusions]{name=concl, column=4, below=r5}{
\begin{itemize}
  \item Eftedal et al.'s low $h^2$ comes mostly from \textbf{how the model was fit}, not from in-laws
  \item With corrected fitting, genetics remains a large source of family resemblance
  \item A family-wide environment is supported out of sample
  \item Additive vs.\ GxG vs.\ family environment \textbf{cannot be cleanly separated} with relative correlations alone
\end{itemize}
}
