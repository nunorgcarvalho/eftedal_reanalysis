% column 6: key above takeaways
\posterbox[adjusted title=Takeaways, fontupper=\fontsize{29}{36}\selectfont]{name=take, column=6, above=bottom}{
\begin{itemize}
  \item Revising Eftedal et al.'s estimator for the Additive model yields \textbf{high $V_A$}
        estimates that explain different relative types better
  \item Additive, GxG and family-environment components are \textbf{collinear}: weak evidence for
        significant components beyond the Additive model
  \item No single model fully explains correlations across all relative types
\end{itemize}
}

\begin{minipage}[t]{0.485\linewidth}
\subhead{Problems in Eftedal et al.'s estimator}
\begin{itemize}
  \item Only optimized for Marginal Pair, not Marginal Type
  \item Log-transformed correlations: more weight to distant relatives
  \item Incorrect inverse-variance weights: more weight to distant relatives
  \item Incorrect expected correlation for step-siblings
  \item Phenotypic ($r$) and genetic ($m$) correlations between mates estimated independently,
        although the model implies $m = V_A\,r$

\end{itemize}

% columns 3-5: models with GxG and family environment
\posterbox[adjusted title=Models with GxG and family environment]{name=gxg, column=3, span=3, between=rel and bottom}{
\begin{minipage}[t]{0.33\linewidth}
\subhead{\color{cGxGText}Gene--gene interactions (GxG)}
Pairwise interactions are shared only when both loci are: sharing decays twice as fast as additive sharing
\[ \rho_k = \dots + V_{\mathrm{GxG}}\left(\frac{1}{4}\right)^{n_k} \]
{\smallsize 0 for in-laws and adoptive relatives\par}

\vspace{1.0em}
\subhead{\color{cFamText}Family environment (F)}
Environment shared across the extended family, decaying with family degree $f$
\[ \rho_k = \dots + V_{\mathrm{fam}}\, r_f^{\,f_k} \]
{\smallsize also shared by in-laws and adoptive relatives; split into an extended part
 ($V_{\mathrm{fam}}\,r_f$, all relatives) and a sib-specific part ($V_{\mathrm{fam}}(1-r_f)$, striped)\par}
